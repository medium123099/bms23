# Положите в КОРЕНЬ сайта как файл .htaccess (если он уже есть — вставьте эти строки в его начало).
# Убирает .html из адресов: /index.html -> /, /catalog.html -> /catalog, /cart.html -> /cart и т.д.

DirectoryIndex index.html

<IfModule mod_rewrite.c>
RewriteEngine On

# 1) /index.html -> /
RewriteCond %{THE_REQUEST} ^[A-Z]{3,9}\s/index\.html[\s?] [NC]
RewriteRule ^index\.html$ / [R=301,L]

# 2) /любая-страница.html -> /любая-страница  (старые ссылки перекинет на новые)
RewriteCond %{THE_REQUEST} ^[A-Z]{3,9}\s/([^\s?]+)\.html[\s?] [NC]
RewriteRule ^ /%1 [R=301,L]

# 3) внутри сервера: /catalog отдаёт файл catalog.html
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME}.html -f
RewriteRule ^([^.]+)$ $1.html [L]
</IfModule>
